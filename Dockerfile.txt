# 1. Use standard Node 20 (Debian-based) instead of Alpine to prevent binary crashes
FROM node:20

# 2. Set the working directory
WORKDIR /app

# 3. Copy package files
COPY package*.json ./

# 4. Install dependencies
# Using --legacy-peer-deps to be safe
RUN npm install --legacy-peer-deps

# 5. Copy the rest of your code
COPY . .

# 6. Build the React Frontend
RUN npm run build

# 7. Set Environment to Production
ENV NODE_ENV=production

# 8. Expose the port
EXPOSE 3000

# 9. Start the server
CMD ["node", "server.js"]
